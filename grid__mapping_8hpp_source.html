36inline float index_to_uv(int i,
37 int n,
38 GridRegistration reg = GridRegistration::CellCenter)
39{
40 if (n <= 0) return 0.f;
41 switch (reg)
42 {
43 case GridRegistration::CellCenter:
44 return (static_cast<float>(i) + 0.5f) / static_cast<float>(n);
45 case GridRegistration::CellCorner:
46 return static_cast<float>(i) / static_cast<float>(n);
47 case GridRegistration::Node:
48 return (n > 1) ? static_cast<float>(i) / static_cast<float>(n - 1) : 0.f;
49 }
50 return 0.f;
51}

56inline float uv_to_continuous_index(
57 float u,
58 int n,
59 GridRegistration reg = GridRegistration::CellCenter)
60{
61 if (n <= 1) return 0.f;
62 float idx = 0.f;
63 switch (reg)
64 {
65 case GridRegistration::CellCenter:
66 idx = u * static_cast<float>(n) - 0.5f;
67 break;
68 case GridRegistration::CellCorner: idx = u * static_cast<float>(n); break;
69 case GridRegistration::Node: idx = u * static_cast<float>(n - 1); break;
70 }
71 return std::clamp(idx, 0.f, static_cast<float>(n - 1));
72}
