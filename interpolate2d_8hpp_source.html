102[[nodiscard]] inline constexpr float bilinear_interp(float f00,
103 float f10,
104 float f01,
105 float f11,
106 float u,
107 float v) noexcept
108{
109 float a10 = f10 - f00;
110 float a01 = f01 - f00;
111 float a11 = f11 - f10 - f01 + f00;
112 return f00 + a10 * u + a01 * v + a11 * u * v;
113}

115[[nodiscard]] inline constexpr float cubic_interpolate(const float p[4],
116 float x) noexcept
117{
118 return p[1] + 0.5f * x *
119 (p[2] - p[0] +
120 x * (2.f * p[0] - 5.f * p[1] + 4.f * p[2] - p[3] +
121 x * (3.f * (p[1] - p[2]) + p[3] - p[0])));
122}
