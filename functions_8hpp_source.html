1289template <typename BaseFn> class GenericFractalFunction : public NoiseFunction
1290{
1291public:
1301 explicit GenericFractalFunction(BaseFn base,
1302 int octaves,
1303 float weight,
1304 float persistence,
1305 float lacunarity);
1306
1312 void set_kw(glm::vec2 new_kw)
1313 {
1314 NoiseFunction::set_kw(new_kw);
1315 this->base.set_kw(new_kw);
1316 }
1317
1323 void set_lacunarity(float new_lacunarity)
1324 {
1325 this->lacunarity = new_lacunarity;
1326 }
1327
1333 void set_octaves(int new_octaves)
1334 {
1335 this->octaves = new_octaves;
1336 this->update_amp0();
1337 }
1338
1344 void set_persistence(float new_persistence)
1345 {
1346 this->persistence = new_persistence;
1347 this->update_amp0();
1348 }
1349
1355 void set_seed(std::uint32_t new_seed)
1356 {
1357 NoiseFunction::set_seed(new_seed);
1358 this->base.set_seed(new_seed);
1359 }
1360
1366 void scale_amp0(float scale)
1367 {
1368 this->amp0 *= scale;
1369 }
1370
1378 void set_stretch(float new_angle,
1379 float new_stretch,
1380 float new_stretch_decay = 0.f)
1381 {
1382 this->angle = new_angle;
1383 this->stretch = new_stretch;
1384 this->stretch_decay = new_stretch_decay;
1385
1386 float theta = new_angle * 3.14159265358979323846f / 180.f;
1387 this->cos_a = std::cos(theta);
1388 this->sin_a = std::sin(theta);
1389 }
1390
1396 float get_lacunarity() const
1397 {
1398 return this->lacunarity;
1399 }
1400
1406 int get_octaves() const
1407 {
1408 return this->octaves;
1409 }
1410
1416 float get_persistence() const
1417 {
1418 return this->persistence;
1419 }
1420
1426 float get_weight() const
1427 {
1428 return this->weight;
1429 }
1430
1431protected:
1436 void update_amp0();
1437
1441 void apply_stretch(float &x, float &y, int octave) const
1442 {
1443 if (this->stretch == 1.f) return;
1444
1445 float decay = std::pow(1.f - std::clamp(this->stretch_decay, 0.f, 1.f),
1446 static_cast<float>(octave));
1447 float stretch_k = 1.f + (this->stretch - 1.f) * decay;
1448 float inv_stretch = 1.f / std::max(stretch_k, 1e-6f);
1449
1450 float xl = x * this->cos_a + y * this->sin_a;
1451 float yl = -x * this->sin_a + y * this->cos_a;
1452 xl *= inv_stretch;
1453 x = xl * this->cos_a - yl * this->sin_a;
1454 y = xl * this->sin_a + yl * this->cos_a;
1455 }
1456
1457protected:
1458 BaseFn base;
1459 int octaves;
1460 float weight;
1461 float persistence;
1462 float lacunarity;
1463 float amp0;
1464 float angle = 0.f;
1465 float stretch = 1.f;
1466 float stretch_decay = 0.f;
1467 float cos_a = 1.f;
1468 float sin_a = 0.f;
1469};
