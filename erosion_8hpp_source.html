1017struct McDonaldParams
1018{
1019 float strength =
1020 0.5f; // overall erosion power (scales suspension & thermal rates)
1021 float deposition = 0.5f; // sediment retention vs transport
1022 float crit_slope = 0.57f; // critical slope [m/m]
1023 float meandering = 0.5f; // flow-coupling & momentum inertia
1024 float scale = 1.0f; // domain extent multiplier (scales world_extent_km and
1025 // base z_scale_km)
1026 float relief_scale = 1.0f; // relative vertical relief scale multiplier
1027
1028 struct PhysicalParams
1029 {
1030 float world_extent_km;
1031 float z_scale_km;
1032 int samples;
1033 int maxage;
1034 float lrate;
1035 float time_step;
1036 float rainfall;
1037 float evap_rate;
1038 float gravity;
1039 float viscosity;
1040 float bed_shear;
1041 float crit_slope;
1042 float settle_rate;
1043 float thermal_rate;
1044 float deposition_rate;
1045 float suspension_rate;
1046 float exit_slope;
1047 };
1048
1049 PhysicalParams to_physical() const
1050 {
1051 PhysicalParams p;
1052 float s = std::max(1e-4f, this->scale);
1053 float r = std::max(1e-4f, this->relief_scale);
1054 p.world_extent_km = 40.f * s;
1055 p.z_scale_km = 4.f * s * r;
1056 p.samples = 8192;
1057 p.maxage = 512;
1058 p.lrate = 0.1f + 0.8f * std::clamp(this->meandering, 0.f, 1.f);
1059 p.time_step = 10.f;
1060 p.rainfall = 1.f;
1061 p.evap_rate = 1e-9f;
1062 p.gravity = 9.81f;
1063 p.viscosity = 0.01f + 0.03f * std::clamp(this->meandering, 0.f, 1.f);
1064 p.bed_shear = 0.02f - 0.015f * std::clamp(this->meandering, 0.f, 1.f);
1065 p.crit_slope = std::max(1e-4f, this->crit_slope);
1066 p.settle_rate = 0.1f *
1067 (0.2f + 1.6f * std::clamp(this->deposition, 0.f, 1.f));
1068 p.thermal_rate = 2.5e-3f *
1069 (0.2f + 1.6f * std::clamp(this->strength, 0.f, 1.f));
1070 p.deposition_rate = 5e-3f *
1071 (0.2f + 1.6f * std::clamp(this->deposition, 0.f, 1.f));
1072 p.suspension_rate = 2.5e-4f *
1073 (0.2f + 1.6f * std::clamp(this->strength, 0.f, 1.f));
1074 p.exit_slope = 0.01f;
1075 return p;
1076 }
1077};
