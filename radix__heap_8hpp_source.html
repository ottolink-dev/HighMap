25// order-preserving mapping between IEEE-754 float and uint32_t for radix

27inline uint32_t float_key(float f)
28{
29 uint32_t u = 0;
30 std::memcpy(&u, &f, 4);

32}

47// monotone priority queue (radix heap) for fast exact flood / Dijkstra
48template <typename T = int> struct RadixHeap
49{
50 std::vector<std::pair<uint32_t, T>> bucket[33];
51 uint32_t last = 0;
52 size_t count = 0;
53
54 bool empty() const
55 {
56 return count == 0;
57 }
58
59 size_t size() const
60 {
61 return count;
62 }
63
64 void clear()
65 {
66 for (auto &b : bucket)
67 b.clear();
68
69 last = 0;
70 count = 0;
71 }
72
73 static int slot(uint32_t x)
74 {
75 return (int)std::bit_width(x);
76 }
77
78 void push(uint32_t key, const T &val)
79 {
80 bucket[slot(key ^ last)].emplace_back(key, val);
81 ++count;
82 }
83
84 void push(uint32_t key, T &&val)
85 {
86 bucket[slot(key ^ last)].emplace_back(key, std::move(val));
87 ++count;
88 }
89
90 std::pair<uint32_t, T> pop()
91 {
92 if (bucket[0].empty())
93 {
94 int i = 1;
95 while (bucket[i].empty())
96 ++i;
97
98 uint32_t mn = 0xffffffffu;
99 for (const auto &e : bucket[i])
100 mn = std::min(mn, e.first);
101
102 last = mn;
103 for (auto &e : bucket[i])
104 bucket[slot(e.first ^ last)].push_back(std::move(e));
105
106 bucket[i].clear();
107 }
108
109 auto e = std::move(bucket[0].back());
110 bucket[0].pop_back();
111 --count;
112 return e;
113 }
114};
