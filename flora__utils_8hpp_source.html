30struct ForestScaleSampler
31{
32 const Array *scale_array = nullptr;
33 glm::vec4 bbox = {0.f, 1.f, 0.f, 1.f};
34 float strength = 1.0f;
35 float bbox_dx = 1.0f;
36 float bbox_dy = 1.0f;
37 bool active = false;
38
39 ForestScaleSampler(const Array &array,
40 float strength_factor,
41 const glm::vec4 &bounds)
42 : scale_array(&array),
43 bbox(bounds),
44 strength(std::clamp(strength_factor, 0.0f, 1.0f)),
45 bbox_dx(bounds.y - bounds.x),
46 bbox_dy(bounds.w - bounds.z)
47 {
48 bool has_data = !array.vector.empty() && validate_non_empty(array);
49 bool valid_bbox = (std::abs(bbox_dx) > 1e-7f && std::abs(bbox_dy) > 1e-7f);
50 active = (has_data && valid_bbox && strength > 0.0f);
51 }
52
53 [[nodiscard]] float sample(float x, float y) const
54 {
55 if (!active || !scale_array) return 1.0f;
56
57 float u_norm = std::clamp((x - bbox.x) / bbox_dx, 0.0f, 1.0f);
58 float v_norm = std::clamp((y - bbox.z) / bbox_dy, 0.0f, 1.0f);
59
60 float xn = u_norm * static_cast<float>(scale_array->shape.x - 1);
61 float yn = v_norm * static_cast<float>(scale_array->shape.y - 1);
62
63 int i = static_cast<int>(xn);
64 int j = static_cast<int>(yn);
65
66 float u = xn - static_cast<float>(i);
67 float v = yn - static_cast<float>(j);
68 float s_raw = std::clamp(scale_array->get_value_bilinear_at(i, j, u, v),
69 0.0f,
70 1.0f);
71 return lerp(1.f, s_raw, strength);
72 }
73};

79struct SpeciesLookup
80{
81 std::unordered_map<uint32_t, Species> map;
82 const std::vector<Species> *list = nullptr;
83
84 explicit SpeciesLookup(const std::vector<Species> &species) : list(&species)
85 {
86 for (const auto &sp : species)
87 {
88 map[sp.id] = sp;
89 }
90 }
91
92 [[nodiscard]] Species get(const Tree &tree) const
93 {
94 auto it = map.find(tree.species_id);
95 if (it != map.end())
96 {
97 return it->second;
98 }
99 if (list && tree.species_id < list->size())
100 {
101 return (*list)[tree.species_id];
102 }
103 return Species(tree.species_id, tree.radius);
104 }
105
106 [[nodiscard]] float get_alpha(const Species &sp_i,
107 size_t s_j,
108 const InteractionMatrix &matrix) const
109 {
110 if (matrix.size > 0)
111 {
112 float mat_val = matrix.get(sp_i.id, s_j);
113 if (mat_val > 0.0f) return mat_val;
114 }
115 return sp_i.competition_factor;
116 }
117};
